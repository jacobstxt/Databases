CREATE TABLE premises (
    premises_id SERIAL PRIMARY KEY,
    premises_type VARCHAR(20) NOT NULL CHECK (premises_type IN ('житлове', 'нежитлове')),
    purpose VARCHAR(30) NOT NULL CHECK (purpose IN ('оренда', 'продаж', 'оренда/продаж')),
    street VARCHAR(150) NOT NULL,
    building_number VARCHAR(20) NOT NULL,
    apartment_number INT,
    entrance_number INT,
    total_area NUMERIC(7, 2) NOT NULL CHECK (total_area > 0),
    living_area NUMERIC(7, 2) CHECK (living_area IS NULL OR living_area <= total_area),
    rooms_count INT NOT NULL DEFAULT 1 CHECK (rooms_count > 0),
    has_kitchen BOOLEAN NOT NULL DEFAULT TRUE,
    bathrooms_count INT NOT NULL DEFAULT 1 CHECK (bathrooms_count >= 0),
    price NUMERIC(12, 2) NOT NULL CHECK (price > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE buyers (
    buyer_id SERIAL PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    email VARCHAR(100),
    bank_account VARCHAR(50) NOT NULL,
    bank_name VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE sellers (
    seller_id SERIAL PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    email VARCHAR(100),
    bank_account VARCHAR(50) NOT NULL,
    bank_name VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);



CREATE TABLE deals (
    deal_id SERIAL PRIMARY KEY,
    premises_id INT NOT NULL REFERENCES premises(premises_id) ON DELETE RESTRICT ON UPDATE CASCADE,
    seller_id INT NOT NULL REFERENCES sellers(seller_id) ON DELETE RESTRICT ON UPDATE CASCADE, 
    buyer_id INT NOT NULL REFERENCES buyers(buyer_id) ON DELETE RESTRICT ON UPDATE CASCADE,
    deal_type VARCHAR(20) NOT NULL CHECK (deal_type IN ('продаж', 'оренда')),
    deal_cost NUMERIC(12, 2) NOT NULL CHECK (deal_cost > 0),
    deal_date DATE NOT NULL DEFAULT CURRENT_DATE,
    rental_term_months INT CHECK (
        (deal_type = 'оренда' AND rental_term_months > 0) OR 
        (deal_type = 'продаж' AND rental_term_months IS NULL)
    ),
    rental_conditions TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
